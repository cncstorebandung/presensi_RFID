<script>
  let adminPin = "";
  let semuaData = [];
  let excelUrl = "";
  let sedangMemuat = false;

  function login() {
    adminPin = document
      .getElementById("pin").value.trim();

    if (!adminPin) {
      document.getElementById("pesan")
        .textContent = "Masukkan PIN dahulu";
      return;
    }

    document.getElementById("pesan")
      .textContent = "Memeriksa PIN...";

    google.script.run
      .withSuccessHandler(function(hasil) {

        document.getElementById("login")
          .classList.add("hidden");

        document.getElementById("dashboard")
          .classList.remove("hidden");

        document.getElementById("tanggal")
          .value = hasil.tanggalHariIni;

        tampilkanData(hasil);

      })
      .withFailureHandler(function(error) {
        document.getElementById("pesan")
          .textContent = error.message;
      })
      .getDashboard(adminPin);
  }

  function tampilkanData(hasil) {
    semuaData = hasil.data;
    excelUrl = hasil.exportUrl;

    document.getElementById("total")
      .textContent = hasil.totalSiswa;

    document.getElementById("hadir")
      .textContent = hasil.hadirHariIni;

    document.getElementById("belum")
      .textContent = hasil.belumHadir;

    document.getElementById("info")
      .textContent =
      "Update terakhir: " +
      new Date().toLocaleTimeString("id-ID");

    renderTabel();
  }

  function renderTabel() {
    const cari = document
      .getElementById("cari")
      .value.toLowerCase();

    const tanggal = document
      .getElementById("tanggal").value;

    const filtered = semuaData
      .filter(r => {
        const cocokNama =
          r.nama.toLowerCase().includes(cari) ||
          r.kelas.toLowerCase().includes(cari);

        const cocokTanggal =
          !tanggal || r.tanggal === tanggal;

        return cocokNama && cocokTanggal;
      })
      .reverse();

    const tbody = document
      .getElementById("isiTabel");

    tbody.replaceChildren();

    if (filtered.length === 0) {
      const tr = document.createElement("tr");
      const td = document.createElement("td");

      td.colSpan = 6;
      td.textContent = "Belum ada data absensi";
      tr.appendChild(td);
      tbody.appendChild(tr);
      return;
    }

    filtered.forEach((r, index) => {

      const tr = document.createElement("tr");

      [
        index + 1,
        r.tanggal,
        r.nama,
        r.kelas,
        r.jam,
        r.status
      ].forEach((value, col) => {

        const td = document.createElement("td");

        if (col === 5) {
          const span = document.createElement("span");
          span.className = "badge";
          span.textContent = value;
          td.appendChild(span);
        } else {
          td.textContent = value;
        }

        tr.appendChild(td);
      });

      tbody.appendChild(tr);
    });
  }

  function resetFilter() {
    document.getElementById("tanggal").value = "";
    document.getElementById("cari").value = "";
    renderTabel();
  }

  function downloadExcel() {
    if (!excelUrl) return;

    // Membutuhkan akses ke file Google Sheets.
    window.open(excelUrl, "_blank");
  }

  function refreshData() {
    if (!adminPin || sedangMemuat) return;

    sedangMemuat = true;

    google.script.run
      .withSuccessHandler(function(hasil) {
        sedangMemuat = false;
        tampilkanData(hasil);
      })
      .withFailureHandler(function(error) {
        sedangMemuat = false;
        document.getElementById("info")
          .textContent = "Gagal refresh: " +
          error.message;
      })
      .getDashboard(adminPin);
  }

  // Refresh setiap 15 detik
  setInterval(refreshData, 15000);
</script>
